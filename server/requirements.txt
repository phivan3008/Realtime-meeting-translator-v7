# --- Web / streaming ---

# --- DSP / array math ---
numpy==2.3.5

soundfile==0.14.0

# --- Torch stack -----------------------------------------------------------
# These three must move together. torchaudio and torchvision are compiled
# against a specific torch ABI, and pip will happily resolve a mismatched
# trio because their metadata does not pin torch tightly enough. A resolver
# run that produced torch 2.13 with torchaudio 2.11 is what put them here as
# explicit pins rather than transitive ones.
torch==2.11.0
torchaudio==2.11.0
torchvision==0.26.0

# --- 1. Voice Activity Detection -------------------------------------------
silero-vad==6.2.1
onnxruntime==1.29.0        # optional ONNX backend for Silero VAD

# --- 2. Deep Noise Filter: AST (AudioSet) ----------------------------------
# See server/pipeline/noise.py for why AST and not YAMNet. In short: YAMNet
# means TensorFlow, TF pins numpy < 2.1 and protobuf 4.x, and everything else
# on this list needs numpy >= 2.
transformers==5.15.1

# --- 3. Overlap Resolver (DSP) ---
pedalboard==0.9.24

# --- 4. Speaker Diarization / voiceprint, and 5. Language ID ---
# Both stages use SpeechBrain checkpoints. Voiceprints are ECAPA-TDNN
# (speechbrain/spkrec-ecapa-voxceleb), language ID is VoxLingua107.
speechbrain==1.1.0

# pyannote.audio is deliberately absent. DESIGN.md names it for this stage
# and it does ship a wrapper around the same ECAPA checkpoint, but that
# wrapper calls SpeechBrain with `token`, `huggingface_cache_dir` and
# `revision`, none of which SpeechBrain 1.1.0 accepts - and pyannote declares
# no version bound on SpeechBrain, so no resolver and no `pip check` can see
# it. server/pipeline/diarization.py calls SpeechBrain directly.

# --- 6. ASR ---
faster-whisper==1.2.1

# --- 7. Translation LLM ---
vllm==0.26.0

# --- Utilities ---
python-dotenv==1.2.3
rich==15.0.0

# ---------------------------------------------------------------------------
# What binds what
#
# vllm is the tightest constraint in the file and decides most of it:
#   vllm >= 0.26  ->  transformers >= 5.5.3   (so transformers 4.x is out)
#   vllm          ->  a specific torch minor
#
# torch then decides torchaudio and torchvision, and numpy >= 2 follows from
# torch 2.11 and vllm alike.
#
# The practical rule: change vllm or torch first, re-resolve the whole file,
# and never pin a single line to fix one error.
#
# To re-resolve after editing:
#   uv pip compile --python-version 3.11 \
#       --python-platform x86_64-unknown-linux-gnu \
#       server/requirements.txt -o server/requirements.lock.txt
